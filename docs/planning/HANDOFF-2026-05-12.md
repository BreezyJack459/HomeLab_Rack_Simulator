**b) `previewRoute` useMemo** (after the `visibleRoutes` useMemo block, before `recommendation`):

```ts
const previewRoute = useMemo((): Route3D | null => {
  if (!previewCable) return null;
  const from = layout.devices.find((d) => d.id === previewCable.fromDeviceId);
  const to   = layout.devices.find((d) => d.id === previewCable.toDeviceId);
  if (!from || !to) return null;
  const plan = calculateCablePlan(previewCable, layout);
  if (!plan) return null;
  const curve = buildCablePath3D(
    previewCable, plan, layout,
    rackWidth, rackDepth, rackHeight,
    0, cableRoutingMode
  );
  if (!curve) return null;
  const meta = CABLE_META[previewCable.type];
  return {
    cable: previewCable,
    plan, from, to, curve,
    color: previewCable.color ?? DEFAULT_CABLE_COLORS[previewCable.type],
    radius: Math.max(meta?.radius ?? 0.008, plan.render.cableRadiusMm / 1000)
  };
}, [previewCable, layout, rackWidth, rackDepth, rackHeight, cableRoutingMode]);
```
